cmake_minimum_required(VERSION 3.27)
project(PyroScreen LANGUAGES C CXX)
if(NOT WIN32)
 message(FATAL_ERROR "PyroScreen currently supports Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(PYROWAVE_UTILS OFF CACHE BOOL "" FORCE)
set(PYROWAVE_DEVEL OFF CACHE BOOL "" FORCE)
set(PYROWAVE_FP32_MATH ON CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_SHADER_MANAGER_RUNTIME_COMPILER OFF CACHE BOOL "" FORCE)
set(GRANITE_SHADER_COMPILER_OPTIMIZE OFF CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_SPIRV_CROSS OFF CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_SYSTEM_HANDLES OFF CACHE BOOL "" FORCE)
set(GRANITE_RENDERER OFF CACHE BOOL "" FORCE)
set(GRANITE_VULKAN_FOSSILIZE OFF CACHE BOOL "" FORCE)
set(GRANITE_PLATFORM "null" CACHE STRING "" FORCE)
set(GRANITE_SHIPPING ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/pyrowave/Granite granite EXCLUDE_FROM_ALL)
add_subdirectory(vendor/pyrowave engine EXCLUDE_FROM_ALL)
# Upstream creates its C shared library only when it is the top-level project.
# Define that same C API here, using unchanged upstream implementation sources.
add_library(pyrowave-shared SHARED
 vendor/pyrowave/pyrowave_c.cpp
 vendor/pyrowave/Granite/video/scaler.cpp
 vendor/pyrowave/pyrowave-shared.def)
target_link_libraries(pyrowave-shared PRIVATE pyrowave granite-vulkan granite-math)
target_compile_definitions(pyrowave-shared PRIVATE PYROWAVE_EXPORT_SYMBOLS)
target_include_directories(pyrowave-shared PRIVATE vendor/pyrowave/shaders vendor/pyrowave/Granite/video)
set_target_properties(pyrowave-shared PROPERTIES PREFIX "lib" OUTPUT_NAME "pyrowave-shared-1")
add_executable(PyroScreen-Sender WIN32 native/pyrowave_2k165_gpu_bench.cpp)
add_executable(PyroScreen-Receiver WIN32 native/pyrowave_udp_client.cpp)
option(PYROSCREEN_TEST_STAGE_TRACE_DEFAULT "Enable per-frame stage diagnostics for development measurements" OFF)
option(PYROSCREEN_TEST_TEARING_DEFAULT "Enable supported vsync-off presentation for development comparisons" OFF)
option(PYROSCREEN_TEST_NVIDIA_DEFAULT "Force NVIDIA receiver adapter for development comparisons" OFF)
option(PYROSCREEN_TEST_FLUSH_PRESENT_DEFAULT "Flush drawing before Present for development comparisons" OFF)
option(PYROSCREEN_TEST_FRAGMENT_DEFAULT "Use fragment shader decoding for development comparisons" OFF)
option(PYROSCREEN_TEST_RGBA_SWAPCHAIN_DEFAULT "Compare RGBA swapchain presentation in development measurements" OFF)
if(PYROSCREEN_TEST_RGBA_SWAPCHAIN_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_RGBA_SWAPCHAIN_DEFAULT)
endif()
if(PYROSCREEN_TEST_FLUSH_PRESENT_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_FLUSH_PRESENT_DEFAULT)
endif()
if(PYROSCREEN_TEST_FRAGMENT_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_FRAGMENT_DEFAULT)
endif()
if(PYROSCREEN_TEST_NVIDIA_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_NVIDIA_DEFAULT)
endif()
if(PYROSCREEN_TEST_TEARING_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_TEARING_DEFAULT)
endif()
if(PYROSCREEN_TEST_STAGE_TRACE_DEFAULT)
 target_compile_definitions(PyroScreen-Sender PRIVATE PYROSCREEN_TEST_STAGE_TRACE_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_STAGE_TRACE_DEFAULT)
endif()
option(PYROSCREEN_TEST_RIO_DEFAULT "Force RIO in sender for development comparisons only" OFF)
if(PYROSCREEN_TEST_RIO_DEFAULT)
 target_compile_definitions(PyroScreen-Sender PRIVATE PYROSCREEN_TEST_RIO_DEFAULT)
endif()
option(PYROSCREEN_TEST_RIO_RECEIVE_DEFAULT "Force RIO in receiver for development comparisons only" OFF)
if(PYROSCREEN_TEST_RIO_RECEIVE_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_RIO_RECEIVE_DEFAULT)
endif()
option(PYROSCREEN_TEST_COALESCED_RECEIVE_DEFAULT "Force UDP receive coalescing for development comparisons only" OFF)
if(PYROSCREEN_TEST_COALESCED_RECEIVE_DEFAULT)
 target_compile_definitions(PyroScreen-Receiver PRIVATE PYROSCREEN_TEST_COALESCED_RECEIVE_DEFAULT)
endif()
target_include_directories(PyroScreen-Sender PRIVATE vendor/pyrowave)
target_include_directories(PyroScreen-Receiver PRIVATE vendor/pyrowave)
target_link_libraries(PyroScreen-Sender PRIVATE pyrowave-shared granite-volk-headers d3d11 d3d12 d3dcompiler dxgi ws2_32 bcrypt crypt32 user32 gdi32)
target_link_libraries(PyroScreen-Receiver PRIVATE pyrowave-shared granite-volk-headers d3d11 d3d12 d3dcompiler dxgi ws2_32 bcrypt crypt32 user32 gdi32)
target_link_options(PyroScreen-Sender PRIVATE /ENTRY:wmainCRTStartup)
target_link_options(PyroScreen-Receiver PRIVATE /ENTRY:wmainCRTStartup)

add_executable(FrameDatagramScheduleTests EXCLUDE_FROM_ALL tests/frame_datagram_schedule_test.cpp)
target_include_directories(FrameDatagramScheduleTests PRIVATE native)
add_executable(DatagramSlotPoolTests EXCLUDE_FROM_ALL tests/datagram_slot_pool_test.cpp)

target_link_libraries(DatagramSenderLoopbackTests PRIVATE granite-volk-headers ws2_32 bcrypt crypt32)
